    <script>
        // APP STATE & ELEMENTS
        let currentHorizon = 6;
        let chartInstance = null;

        const inputs = {
            initialCapital: document.getElementById('initialCapital'),
            evalPrice: document.getElementById('evalPrice'),
            winRate: document.getElementById('winRate'),
            targetPayout: document.getElementById('targetPayout'),
            reinvestBudget: document.getElementById('reinvestBudget'),
            investorShare: document.getElementById('investorShare')
        };

        const displays = {
            winRateVal: document.getElementById('winRateVal'),
            investorShareVal: document.getElementById('investorShareVal'),
            kpiInitial: document.getElementById('kpiInitial'),
            kpiActiveFunded: document.getElementById('kpiActiveFunded'),
            kpiInvestorProfit: document.getElementById('kpiInvestorProfit'),
            kpiInvestorROI: document.getElementById('kpiInvestorROI'),
            kpiTraderProfit: document.getElementById('kpiTraderProfit'),
            tableBody: document.getElementById('financialTableBody')
        };

        // EVENT LISTENERS
        Object.values(inputs).forEach(input => {
            input.addEventListener('input', calculateAndRender);
        });

        document.querySelectorAll('.horizon-btn').forEach(btn => {
            btn.addEventListener('click', (e) => {
                document.querySelectorAll('.horizon-btn').forEach(b => {
                    b.classList.remove('border-brand-500', 'bg-brand-500/20', 'text-brand-300');
                    b.classList.add('border-slate-700', 'bg-slate-800', 'text-slate-300');
                });
                e.target.classList.remove('border-slate-700', 'bg-slate-800', 'text-slate-300');
                e.target.classList.add('border-brand-500', 'bg-brand-500/20', 'text-brand-300');
                currentHorizon = parseInt(e.target.dataset.months);
                calculateAndRender();
            });
        });

        // CORE CALCULATION ENGINE
        function calculateAndRender() {
            const initialCap = parseFloat(inputs.initialCapital.value) || 1000;
            const price = parseFloat(inputs.evalPrice.value) || 135;
            const winRatePct = parseFloat(inputs.winRate.value) || 33.3;
            const winRateDecimal = winRatePct / 100;
            const payoutPerFunded = parseFloat(inputs.targetPayout.value) || 2000;
            const reinvestVal = parseFloat(inputs.reinvestBudget.value) || 1000;
            const invSharePct = parseFloat(inputs.investorShare.value) || 50;

            displays.winRateVal.innerText = `${winRatePct.toFixed(1)}% (1 из ${(100/winRatePct).toFixed(1)})`;
            displays.investorShareVal.innerText = `${invSharePct}%`;

            let tableHTML = '';
            let cumulativeActiveFunded = 0;
            let totalInvestorReturn = 0;
            let totalTraderReturn = 0;

            const labels = [];
            const grossData = [];
            const investorData = [];
            const traderData = [];

            for (let m = 1; m <= currentHorizon; m++) {
                labels.push(`Месяц ${m}`);

                let evalsBought = 0;
                let opex = 0;
                let reinvest = 0;

                if (m === 1) {
                    evalsBought = 3;
                    opex = 595;
                    reinvest = 1000;
                } else {
                    reinvest = reinvestVal;
                    evalsBought = Math.floor(reinvest / price);
                    opex = 1000;
                }

                // Passed accounts calculation
                const passedThisMonth = Math.max(1, Math.round(evalsBought * winRateDecimal));
                cumulativeActiveFunded += passedThisMonth;

                // Gross payout
                const grossPayout = cumulativeActiveFunded * payoutPerFunded;

                // Net Business Profit
                const netProfit = Math.max(0, grossPayout - reinvest - opex);

                // Investor & Trader Share
                let investorShare = 0;
                let traderShare = 0;

                if (m <= 3) {
                    // Grace Period: 100% stays in business / trader
                    investorShare = 0;
                    traderShare = netProfit;
                } else {
                    investorShare = netProfit * (invSharePct / 100);
                    traderShare = netProfit * ((100 - invSharePct) / 100);
                }

                totalInvestorReturn += investorShare;
                totalTraderReturn += traderShare;

                grossData.push(grossPayout);
                investorData.push(investorShare);
                traderData.push(traderShare);

                tableHTML += `
                    <tr class="hover:bg-slate-800/40 transition-colors">
                        <td class="p-3 font-semibold text-white">M${m}</td>
                        <td class="p-3">${evalsBought}</td>
                        <td class="p-3 text-emerald-400">+${passedThisMonth}</td>
                        <td class="p-3 font-bold text-brand-400">${cumulativeActiveFunded}</td>
                        <td class="p-3">$${grossPayout.toLocaleString()}</td>
                        <td class="p-3">$${reinvest.toLocaleString()}</td>
                        <td class="p-3">$${opex.toLocaleString()}</td>
                        <td class="p-3 font-medium text-white">$${netProfit.toLocaleString()}</td>
                        <td class="p-3 font-bold text-emerald-400">$${investorShare.toLocaleString()}</td>
                        <td class="p-3 font-bold text-sky-400">$${traderShare.toLocaleString()}</td>
                    </tr>
                `;
            }

            displays.tableBody.innerHTML = tableHTML;

            // UPDATE KPIs
            displays.kpiInitial.innerText = `$${initialCap.toLocaleString()}`;
            displays.kpiActiveFunded.innerText = cumulativeActiveFunded;
            displays.kpiInvestorProfit.innerText = `$${totalInvestorReturn.toLocaleString()}`;
            
            const roi = (totalInvestorReturn / initialCap) * 100;
            displays.kpiInvestorROI.innerText = `ROI: ${roi.toFixed(0)}%`;
            displays.kpiTraderProfit.innerText = `$${totalTraderReturn.toLocaleString()}`;

            // UPDATE CHART
